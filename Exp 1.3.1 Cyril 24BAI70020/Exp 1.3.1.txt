class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


class Solution:
    def isPalindrome(self, head):
        if head is None or head.next is None:
            return True

        slow = head
        fast = head

        while fast and fast.next:
            slow = slow.next
            fast = fast.next.next

        prev = None
        curr = slow

        while curr:
            nxt = curr.next
            curr.next = prev
            prev = curr
            curr = nxt

        left = head
        right = prev
        result = True

        while right:
            if left.val != right.val:
                result = False
                break
            left = left.next
            right = right.next

        curr = prev
        prev = None

        while curr:
            nxt = curr.next
            curr.next = prev
            prev = curr
            curr = nxt

        return result


def create_linked_list(values):
    head = None
    tail = None

    for value in values:
        new_node = ListNode(value)

        if head is None:
            head = new_node
            tail = new_node
        else:
            tail.next = new_node
            tail = new_node

    return head


def display_linked_list(head):
    values = []

    while head:
        values.append(str(head.val))
        head = head.next

    print(" → ".join(values))

head = create_linked_list(values)
