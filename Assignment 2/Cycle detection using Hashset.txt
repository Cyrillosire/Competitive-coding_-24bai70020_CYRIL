class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


def hasCycle(head):
    visited = set()

    while head:
        if head in visited:
            return True

        visited.add(head)
        head = head.next

    return False


def main():
    values = list(map(int, input("Enter linked list values: ").split()))
    pos = int(input("Enter cycle position (-1 for no cycle): "))

    if not values:
        print("Cycle detected:", False)
        return

    nodes = [ListNode(value) for value in values]

    for i in range(len(nodes) - 1):
        nodes[i].next = nodes[i + 1]

    if 0 <= pos < len(nodes):
        nodes[-1].next = nodes[pos]

    print("Cycle detected:", hasCycle(nodes[0]))


if __name__ == "__main__":
    main()
