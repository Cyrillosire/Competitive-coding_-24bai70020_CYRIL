class Solution:
    def containsNearbyDuplicate(self, nums, k):
        window = set()

        for i in range(len(nums)):
            if nums[i] in window:
                return True

            window.add(nums[i])

            if len(window) > k:
                window.remove(nums[i - k])

        return False


nums = [1, 0, 1, 1]
k = 1

sol = Solution()
print(sol.containsNearbyDuplicate(nums, k))
