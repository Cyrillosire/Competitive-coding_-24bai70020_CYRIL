class Solution:
    def containsNearbyDuplicate(self, nums, k):
        seen = {}

        for i, num in enumerate(nums):
            if num in seen and i - seen[num] <= k:
                return True

            seen[num] = i

        return False



nums = [1, 0, 2, 1]
k = 2

sol = Solution()
print(sol.containsNearbyDuplicate(nums, k))
