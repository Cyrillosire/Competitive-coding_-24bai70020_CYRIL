class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


def detectCycle(head):
    slow = head
    fast = head

    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next

        if slow == fast:
            slow = head

            while slow != fast:
                slow = slow.next
                fast = fast.next

            return slow

    return None


def main():
    values = list(map(int, input("Enter linked list values: ").split()))
    pos = int(input("Enter cycle position (-1 for no cycle): "))

    if not values:
        print("Cycle start:", None)
        return

    nodes = [ListNode(value) for value in values]

    for i in range(len(nodes) - 1):
        nodes[i].next = nodes[i + 1]

    if 0 <= pos < len(nodes):
        nodes[-1].next = nodes[pos]

    cycle_start = detectCycle(nodes[0])

    if cycle_start:
        print("Cycle starts at node with value:", cycle_start.val)
    else:
        print("No cycle")


if __name__ == "__main__":
    main()
